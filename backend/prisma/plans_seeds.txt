const { PrismaClient } = require('@prisma/client');

const prisma = new PrismaClient();

// Helper to build the features array from a plan object
function buildFeatures(plan, isSeasonal = false) {
  const features = [];

  // AI scans
  features.push(`${plan.aiScansPerDay} AI scan${plan.aiScansPerDay !== 1 ? 's' : ''} per day`);

  // Consultations
  if (plan.consultationsPerMonth > 0) {
    features.push(`${plan.consultationsPerMonth} consultation${plan.consultationsPerMonth !== 1 ? 's' : ''} per month`);
  } else {
    features.push('No consultations');
  }

  // Meal plan type
  if (plan.mealPlanType) {
    features.push(plan.mealPlanType + 'meal plan');
  }

  // Recipe library
  if (plan.recipeLibrary) features.push('Full recipe library');

  // Food diary
  if (plan.foodDiary) features.push('Food diary & tracking');

  // Chatbot
  if (plan.chatbot) features.push('AI chatbot assistant');

  // For seasonal plans, add duration as a feature
  if (isSeasonal && plan.duration) {
    features.push(`Valid for ${plan.duration}`);
  }

  return features;
}
