import { useEffect, useState } from "react";
import { nutritionistPatientsApi } from "@/services/api";
import type { PatientProfile } from "@/types/api";

export default function PatientProfiles() {
  const [patients, setPatients] = useState<PatientProfile[]>([]);
  const [loading, setLoading] = useState(true);
  const [selected, setSelected] = useState<PatientProfile | null>(null);

  useEffect(() => {
    nutritionistPatientsApi
      .my()
      .then((r) => {
        const list = r.patients ?? [];
        setPatients(list);
        if (list[0]) setSelected(list[0]);
      })
      .finally(() => setLoading(false));
  }, []);

  if (loading) {
    return <div className="kl-card text-center py-8">Loading patient profiles…</div>;
  }

  if (patients.length === 0) {
    return (
      <div className="kl-card text-center py-12">
        <i className="fas fa-id-card text-3xl text-muted-foreground mb-2 block" />
        <p className="text-muted-foreground">No patient profiles yet.</p>
      </div>
    );
  }

  // Use nested user fields
  const name = (p: PatientProfile) => p.user.fullName ?? `Patient #${p.id}`;
  const email = (p: PatientProfile) => p.user.email ?? "";

  return (
    <div className="grid grid-cols-1 md:grid-cols-[300px_1fr] gap-5">
      {/* Sidebar list */}
      <div className="kl-card p-0 max-h-[calc(100vh-200px)] overflow-y-auto">
        {patients.map((p) => {
          const isActive = selected?.id === p.id;
          return (
            <button
              key={p.id}
              onClick={() => setSelected(p)}
              className={`w-full text-left flex items-center gap-3 px-4 py-3 border-b border-gray-line transition-colors ${
                isActive ? "bg-cream-bg border-l-4 border-l-kl-orange" : "hover:bg-cream-bg"
              }`}
            >
              <div className="sb-av w-9 h-9">{name(p)[0]?.toUpperCase()}</div>
              <div className="min-w-0">
                <div className="font-syne font-semibold text-sm">{name(p)}</div>
                <div className="text-xs text-muted-foreground">{email(p)}</div>
              </div>
            </button>
          );
        })}
      </div>

      {/* Selected patient details */}
      {selected && (
        <div className="kl-card">
          <div className="kl-card-head">
            <div className="kl-card-title">
              <i className="fas fa-id-card" /> {name(selected)}
            </div>
          </div>
          <div className="grid grid-cols-2 md:grid-cols-3 gap-3">
            <Stat label="Email" value={email(selected) || "—"} />
            <Stat label="Age" value={selected.age != null ? String(selected.age) : "—"} />
            <Stat label="Height" value={selected.height != null ? `${selected.height} cm` : "—"} />
            <Stat label="Weight" value={selected.weight != null ? `${selected.weight} kg` : "—"} />
            <Stat label="Goal Weight" value={selected.goalWeight != null ? `${selected.goalWeight} kg` : "—"} />
          </div>
          {selected.goals?.length ? <Block title="Goals" icon="bullseye" items={selected.goals} /> : null}
          {selected.allergies?.length ? <Block title="Allergies" icon="triangle-exclamation" items={selected.allergies} /> : null}
          {/* Notes removed – not in backend schema. Add if you add notes to Patient model later. */}
        </div>
      )}
    </div>
  );
}

function Stat({ label, value }: { label: string; value: string }) {
  return (
    <div className="bg-cream-bg p-3 rounded-xl">
      <div className="text-[0.65rem] uppercase tracking-wider text-muted-foreground">{label}</div>
      <div className="font-syne font-bold text-base">{value}</div>
    </div>
  );
}

function Block({ title, icon, items }: { title: string; icon: string; items: string[] }) {
  return (
    <div className="mt-3">
      <div className="font-syne font-bold text-sm mb-1">
        <i className={`fas fa-${icon} text-kl-orange mr-1`} /> {title}
      </div>
      <div className="flex flex-wrap gap-1">
        {items.map((it, i) => (
          <span key={i} className="kl-tag kl-tag-gray text-xs">
            {it}
          </span>
        ))}
      </div>
    </div>
  );
}
